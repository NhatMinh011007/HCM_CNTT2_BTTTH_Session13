* Nội dung file trong scripts/:
* PowerShell

$date = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
"BAO CAO SU CO SMARTFARM - $date" > ..\backup\bao-cao-su-co.txt
"" >> ..\backup\bao-cao-su-co.txt


$logs = Get-Content ..\data\log-he-thong.log | Select-String "CRITICAL","ERROR"
$logs | ForEach-Object { $_.Line } >> ..\backup\bao-cao-su-co.txt


$count = ($logs | Measure-Object).Count
"" >> ..\backup\bao-cao-su-co.txt
"TONG SO SU CO: $count" >> ..\backup\bao-cao-su-co.txt
